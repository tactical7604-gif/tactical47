	function showConncTimeTopMenu() {
		if($('#topUserLoginTime').contents().length == 0) return;
		var lastLoginTime = localStorage.getItem('lastLoginTime');
		
		var second = 1000;
		var minute = second * 60;
		var hour = minute * 60;
		var day = hour * 24;
		
		setInterval(function() {
			
			var time = new Date().getTime() - lastLoginTime
	    	var days = Math.floor(time / day);
			var hours = Math.floor((time % day) / hour);
			var minutes = Math.floor((time % hour) / minute);
			var seconds = Math.floor((time % minute) / second);
			
			if(hours < 10) hours = "0" + hours;
			if(seconds < 10) seconds = "0" + seconds;
			if(minutes < 10) minutes = "0" + minutes;
			
			// 건전 팝업 안내
			if(isLogin && hours == 1 && minutes == 0 && seconds == 0)	popupOpen();		// 60분(1시간) 실행
			if(isLogin && hours == 1 && minutes == 30 && seconds == 0)	popup90Open();	// 90분(1시간 30분) 실행

			$('#topUserLoginTime').contents().filter(function() {
			    if(this.nodeType === 3) {
					if(time > 0) {
						if(time >= hour) {
							this.textContent = hours + ":" + minutes + ":" + seconds;
						} else {
							this.textContent = minutes + ":" + seconds;
						}	
					} else {
						this.textContent = "00:00";
					}
			    }
			});
		}, second);
	}

	function initTouchClickHandlerBannerTopMenu() {
		var touchStartX = 0;
		var touchStartY = 0;
		var touchMoved = false;
		
		$('.slide-menu-banner.bxSlider')
			.off('touchstart touchmove touchend click')
			.on('touchstart', function(e) {
				touchMoved = false;
				var touch = e.originalEvent.touches[0];
				touchStartX = touch.pageX;
				touchStartY = touch.pageY;
			})
			.on('touchmove', function(e) {
				var touch = e.originalEvent.touches[0];
				var diffX = Math.abs(touch.pageX - touchStartX);
				var diffY = Math.abs(touch.pageY - touchStartY);
				if (diffX > 10 || diffY > 10) touchMoved = true;
			})
			.on('touchend', function(e) {
				if(!touchMoved) {
					var target = $(e.target).closest('a');
					if (target.length) {
						e.preventDefault();
						location.href = target.attr('href');
					}
				}
			});
	}

	function checkImagesLoadedTopMenu(obj, callback) {
		var $imgs = obj.find('img');
		var total = $imgs.length;
		var loaded = 0;

		$imgs.each(function () {
			if (this.completh) {
				loaded++;
			} else {
				$(this).on('load', function () {
					loaded++;
					if (loaded === total) {
						callback();
					}
				});
			}
		});
		
		//캐시로 인해 모든 이미지가 로딩된 경우
		if (loaded === total) {
			callback();
		}	
	}

/**

 * @param markingList 마킹정보
 * @param scheduleList 게임정보
 */
rightA.setRightDataProtoGame = function(markingList, scheduleList) {
	// 카트 구매하기 영역 갱신
	var totalCnt = protoVictory.getSlipTotSelectCnt();
	var inputPrice = Number($('#txtBettingMoney').val().replace(/,/g, ''));
	var predictionWinTotal = 0;
	//inputPrice = inputPrice ? inputPrice : slipUtil.baseSellPrice;
	var totalBuyAmt = 0;//[신규추가]
	var buyAmtCtt = "";//[신규추가] "1000,2000,3000"

	var data = {
		gmId : protoVictory.gmId
		,totSelectCnt : totalCnt
		,selectPrice : _slipConfig.gameDivCd == 'S' ? '-' : inputPrice
		,totPrice : totalCnt ? inputPrice * totalCnt : '-'
		,availableBuyAmount : memoryScheduleInfo.availableBuyAmount
		,currentLottery : memoryScheduleInfo.currentLottery
		,gameTs : protoVictory.gmTs
		,slmkCtt : protoVictory.makeSlmkCtt(inputPrice)
		,buyAmt : totalCnt ? buyAmtCtt : '-'
		,totalBuyAmt : totalBuyAmt
	};
	
	//console.log(data);
	
	if(data.totSelectCnt) {
		$("#divCartData").show();
		$("#divCartNoData").hide();	
		
		// 예상 배당률 처리
		var allot = 1;
		var html = '';
		for(var matchSeq in markingList) {
			var rdata = {};
			var prlYnHtml = '';
			var selKey = markingList[matchSeq];
			selKey = Number(selKey);
			var row = scheduleList[matchSeq];
			row.allot = selKey === 1 ? row.winAllot : selKey === 2 ? row.drawAllot : row.loseAllot;
			allot *= row.allot*1000;
			if(row.prlYn == 'Y') {
				prlYnHtml = '<i class="ico-parlay"><span class="blind">Parlay</span></i>';
			}
			rdata.prlYnHtml = prlYnHtml;
			// 콤마세팅해서 배팅액 입력하기(한경기 입력금액)
			var minAmt = 1000;
			if (row.handi == '12' || row.handi == '24') {
				minAmt = 2000;
			}			
			var betAmount = row.betAmount === undefined ? betmanOnlineFront.setNumberComma(minAmt) : betmanOnlineFront.setNumberComma(row.betAmount.replace(/[^0-9]/g, ''));
			// 승부식 한경기 개별 금액이 있을 경우 해당 데이터 셋팅
			if (_slipConfig.gameDivCd == 'S') {
				betAmount = (matchSeq in prevInputValues) ? prevInputValues[matchSeq] : betAmount;
			}			

			switch(selKey) {
			case 1:
				rdata.txt = row.winTxt;
				break;
			case 2:
				rdata.txt = row.drawTxt;
				break;
			case 3:
				rdata.txt = row.loseTxt;
				break;
			default:
				rdata.txt = "신규";
			}
			
			if (rdata.txt == "ODD")
				rdata.txt = "홀";
			else if (rdata.txt == "EVEN")
				rdata.txt = "짝";
			else if (rdata.txt == "1")

					rightA.modalID = BUICommon.openModalByObj($(this), "#changedAllotConfirmDlg", '배당률 및 기준값 변경안내', '', false, null);

					$("#changedAllotConfirmBtn2").on('click', function() { rightA.closeChangedAllotCartPopup(true); });
					//$("div[name='changedAllotConfirmDlg']").find(".popClose").on('click', function() { rightA.closeChangedAllotCartPopup(false); });
				}
			}
			
			//최신정보 조회 및 refresh
			if(rightA.cartData.gmId == 'G101' || rightA.cartData.gmId == 'G104') {
				rightA.protoVictoryChangeReload(popupCallback);
			}
			return false;
			
		}else{
			if(result.protoErrorMessage){
				if(result.protoErrorMessage.indexOf("취소된 경기가 있습니다") > -1) {
					alert("취소된 경기가 있습니다. \n해당 경기를 삭제하고 카트에 담으실 수 있습니다.");
				} else if(result.protoErrorMessage.indexOf('발매가 일시 중지 되었습니다') > -1) {
					alert(getMessage('BF0065')); //발매 차단 경기가 있습니다. \n해당 경기를 삭제하고 카트에 담으실 수 있습니다.
				} else {
					alert(betmanOnlineFront.parseAlertMsg(result.protoErrorMessage));
					if(result.protoErrorMessage.indexOf('강제탈퇴 되었습니다') > -1) forceLogout();
				}
				//최신정보 조회 및 refresh
				if(rightA.cartData.gmId == 'G101' || rightA.cartData.gmId == 'G104') {
					rightA.protoVictoryChangeReload(function(){});
				}
				return false;
				
			} else {
				rightA.insertToCart();
				return false;
			}
		}
	}
	return true;
}

/**
 * right 카트담기 validation
 */
rightA.valid = function(flag) {
	var gmId = this.cartData.gmId;
	var slmkCtt = this.cartData.slmkCtt;
	var totPrice = this.cartData.totPrice;

	var buyAmt = 0;
	var totalAmt = 0;
	var selectPrice;
	var selectPriceArr = [];
	if(this.cartData.selectPrice && this.cartData.selectPrice != '-') {
		selectPrice = this.cartData.selectPrice;
	} else {
		selectPrice = this.cartData.buyAmt;
	}
	if(this.cartData.buyAmt) {
		buyAmt = this.cartData.buyAmt;
	} 
	if(this.cartData.totPrice) {
		totalAmt = this.cartData.totPrice;
	} else {
		totalAmt = buyAmt;
	}

	if (_slipConfig.gameDivCd == 'S') {
		if (selectPrice.indexOf(",") > -1) {
			selectPriceArr = selectPrice.split(",");
		} else {
			selectPriceArr = [selectPrice];
		}
		
	}
	
	var availableBuyAmount = this.cartData.availableBuyAmount;
	var msgCd = null;
	var minAmt = slipUtil.unitSellPrice;
	
	//모의체험인 경우에는 5만원 구매가능 금액 체크 하지 않도록 flag를 둠
	var chkAmount = true;
	if (typeof flag != 'undefined' && flag == 'mockJoin') {
		chkAmount = false;
	}
	
	switch (gmId) {
	case 'G101':
	case 'G104':
		// G101, G104 공통 처리	
		var gameCount = slmkCtt.replace(/^([0-9]+).*$/,'$1');
		if(_slipConfig.gameDivCd != 'S' && (gameCount < 2 || 10 < gameCount)) {
			msgCd = 'BF0060';	// 승부식 게임은 2~10경기까지 구매가능합니다.
		}
		
		// 발매차단 게임취소 등 상태 check 추가
		for(var matchSeq in markingList) {
			var game = _slipConfig.data.orgScheduleList[matchSeq];
			var selKey = markingList[matchSeq];
			
			//취소 경기 확인
			if (game.winAllot == 1 || game.gameResult == '4') {
				if(rightA.directBuy == 'Y') { //메세지 추후 공통메시지로 관리 필요
					//msg = '취소된 경기가 있습니다. \n해당 경기를 삭제하고 구매페이지로 이동하실 수 있습니다.';
					msgCd = 'BF0178';
				} else {
					//msg = '취소된 경기가 있습니다. \n해당 경기를 삭제하고 카트에 담으실 수 있습니다.';	
					msgCd = 'BF0177';
				}
			}

			// 마감 경기 확인
			if(game.protoStatus != '2') {
				if(rightA.directBuy == 'Y') {
					msgCd = 'BF0074';
				} else {
					msgCd = 'BF0064';
				}
			}
			
			// 차단 경기 확인
			var isReject = false;
			if(selKey == 1 && game.gameReject == '1' && ['1', '3', '5', '7'].indexOf(game.buyReject) > -1) isReject = true;
			if(selKey == 2 && game.gameReject == '1' && ['2', '3', '6', '7'].indexOf(game.buyReject) > -1) isReject = true;
			if(selKey == 3 && game.gameReject == '1' && ['4', '5', '6', '7'].indexOf(game.buyReject) > -1) isReject = true;

			if(isReject) {
				if(rightA.directBuy == 'Y') {
					msgCd = 'BF0075';
				} else {
					msgCd = 'BF0065';	
				}	
			}
			
			// 게임(조합 C/) 상태 추가
			if (_slipConfig.gameDivCd == 'C' && game.sgl == '2') {
				alert('한경기 전용 상품이 포함되어 있습니다.')
				return true;
			}
			
			// 게임(한경기 S) 상태 추가
			if (_slipConfig.gameDivCd == 'S' && game.sgl == '0') {
				alert('조합 전용 상품이 포함되어 있습니다.')
				return true;
			}			
		}

		if (_slipConfig.gameDivCd == 'S') {
			var slmkCttArr = [];
			var priceCheckArr = [];
			
			slmkCttArr = slmkCtt.split(",");
			for(var i=1; i < slmkCttArr.length - 2; i++){
				$.each(_slipConfig.data.orgScheduleList, function(key, val){
					if (val.matchSeq == slmkCttArr[i].substr(0,3) + slmkCttArr[i].substr(4,1)) {
						priceCheckArr.push({matchSeq : val.matchSeq, handi : val.handi});
					}
				});
			}
			
			for(var ii in priceCheckArr) {
				if (priceCheckArr[ii].handi == '12' || priceCheckArr[ii].handi == '24') {
					if ((selectPriceArr[ii] % 2000 > 0) || (selectPriceArr[ii] < 2000)) {
						alert('개별투표금액은 2000원 단위로 입력해 주세요.')
						return true;
					}
				}
			}
		}
		
		//적중예상금액 1억 초과
		if(totPrice > 100000000){
			alert('투표권당 예상 적중금은 1억원을 초과할 수 없습니다.');
			return true;
		}
		
		// 프로토 예외 처리
		totPrice = selectPrice;
		
		// 싱글일 경우 구매 최소 금액(1,000원) 변경 처리
		if(_slipConfig.gameDivCd == 'S') {
			minAmt = 1000;
		}
		
		break;
	case 'G102':
		$('#buyList [name="buyAmt"]').each(function(){
			var amt = $(this).val().replace(',' ,'')
			if(amt <= 0) {
				msgCd = 'BF0112' // 개별투표금액을 입력해 주세요.
				return false;
			} else if(amt % minAmt > 0) {
				msgCd = 'BF0045' // 개별투표금액은 {게임별 최소단위 금액}원 단위로 입력해 주세요.
				return false;
			}
		});
		break;
	}

	if(!msgCd && msgCd != 'msg') {
		if(totalAmt <= 0) {
			msgCd = 'BF0112'; // 개별투표금액을 입력해 주세요.
		} else if(selectPrice % minAmt > 0) {
			msgCd = 'BF0045'; // 개별투표금액은 {게임별 최소단위 금액}원 단위로 입력해 주세요.
		} else if (selectPriceArr) {
			for(var i=0; i < selectPriceArr.length; i++){
				if(selectPriceArr[i] % minAmt > 0) {
					msgCd = 'BF0045'; // 개별투표금액은 {게임별 최소단위 금액}원 단위로 입력해 주세요.
				}
				if(selectPriceArr[i] <= 0) {
					msgCd = 'BF0112'; // 개별투표금액을 입력해 주세요.
				}
			}
		} else if(totalAmt > availableBuyAmount && rightA.directBuy != 'N') {//카트담기는 가능
			msgCd = 'BF0068'; // 회차별 10만원까지 구매가 가능합니다. \r\n 구매가능금액을 확인해 주세요.
		}
	}
	
	var msg;
	
	if(msgCd && msgCd != 'msg') {
		if(msgCd == "BF0045") {
			msg = getMessage(msgCd , betmanOnlineFront.setNumberComma(minAmt));
		} else {
			msg = getMessage(msgCd);
		}
	}
	
	if(msg) {
		alert(msg);
		return true;
	}
	return false;
}

 * 카트 최대 유효성검사
 */
rightA.maxCartValid = function(){
	var gmId = this.cartData.gmId;
	var gameType = [];
	var msgCd = null;
	var totalCount = -1;
	if(gmId == 'G102') {
		$("#buyList li").each(function(){
			if(gameType.indexOf($(this).data("gamealphaid")) == -1) {
				gameType.push($(this).data("gamealphaid"));	
			}
		});
	}
	
	if(gameType.length == 0) gameType.push('A');
	
	requestClient.requestPostMethod("/main/inqCartGameListCnt.do", {}, false, function(result){
		totalCount = result.items.totalCnt;
	}, rightA.errorCallback);
	
	if(totalCount + gameType.length > 100) {
		msgCd = 'BF0063'; // 쇼핑카트는 현재 100개까지 담으실 수 있습니다. \r\n [현재 카트 보관수 : {0} 개]
	}
	
	if(msgCd) {
		alert(getMessage(msgCd, totalCount));
		return true;
	}
	if(totalCount == -1) {
		return true;
	}
	return false;
}

rightA.checkCartCallback = function(result){
	if(result){
		if(result.protoChangeMessage){

				var popupCallback = function() {};
			} else {

				rightA.checkCartPopup(result);

				}
			}
			//console.log("rightA.checkCartCallback:: rightA.protoRecordChangeReload 수행전.");    //TODO:삭제필요
			//최신정보 조회 및 refresh
			if(rightA.cartData.gmId == 'G101' || rightA.cartData.gmId == 'G104') {

			}
			//console.log("rightA.checkCartCallback:: rightA.protoRecordChangeReload 수행 후.");    //TODO:삭제필요
			return false;
			
		}else{
			if(result.protoErrorMessage){
				if(result.protoErrorMessage.indexOf("취소된 경기가 있습니다") > -1) {
					alert("취소된 경기가 있습니다. \n해당 경기를 삭제하고 구매페이지로 이동하실 수 있습니다.");
				} else if(result.protoErrorMessage.indexOf('발매가 일시 중지 되었습니다') > -1) {
					alert(getMessage('BF0075')); //발매 차단 경기가 있습니다. \n해당 경기를 삭제하고 카트에 담으실 수 있습니다.
				} else {
					alert(betmanOnlineFront.parseAlertMsg(result.protoErrorMessage));

				}
				return false;
				
			}
		}
	}
	//console.log("rightA.checkCartCallback:: directBuy=Y : rightA.insertToDirectCart 수행 전.");
	if(rightA.directBuy == 'Y') {
		if(confirm(getMessage('BF0083'))) {
			rightA.insertToDirectCart();
		} else {
			return false;
		}
	} else {

	}	
}

String.prototype.odds = function() {
	var o =  (this/100).toString();
	if(o.indexOf(".") == -1){
		o += ".00";
	}else{
		if(o.substring(o.indexOf(".")+1).length == 1)
			o += "0";
	}
	return o;
}

//[HANDI] 핸디값변경시 수정필요
String.prototype.vic = function(handi) {
	var list = _slipConfig.data.orgData.schedulesList.filter(function(row){
		return handi.indexOf(row.matchSeq) > -1;
	});
	
	if (list.length > 0) {
		var obj = list[0];
		var txt = "";
		if(this == "1"){
			txt = obj.winTxt;
		}else if(this == "2"){
			txt = obj.drawTxt;
		}else if(this == "3"){
			txt = obj.loseTxt;
		}
		if (txt == "ODD")
			txt = "홀";
		else if (txt == "EVEN")
			txt = "짝";
		else if (txt == "1")
			txt = "①";
		else if (txt == "5")
			txt = "⑤";
		return txt;		
	} else {
		if(handi == "9" || handi == "19"){
			if(this == "1"){
				return "언더";
			}else if(this == "2"){
				return "";
			}else if(this == "3"){
				return "오버";
			}
		}else{
			if(this == "1"){
				return "승";
			}else if(this == "2"){
